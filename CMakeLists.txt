cmake_minimum_required(VERSION 3.16)
project(cache_simulator VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(cache_sim src/main.cpp)
add_executable(cache_tests tests/test_cache.cpp)
foreach(target cache_sim cache_tests)
  target_include_directories(${target} PRIVATE include)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME cache_unit COMMAND cache_tests)
